"""Bounded end-to-end actor-critic timing; repeated episodes reuse the scene."""
import argparse,functools,gc,hashlib,io,json,platform,resource,time
from pathlib import Path
import numpy as np
import torch
from voxelform.context import ContextLibrary
from voxelform.environment import Environment
from voxelform.patch_policy import PatchConfig,PatchFacades,PatchLearner
from voxelform.shadows import sync
from voxelform.workload import action_reward

ROOT=Path(__file__).resolve().parents[1]

def episode(e,p,seed,training=True):
    torch.manual_seed(seed);p.clear();times=dict(reset=0.,choose=0.,model=0.,sampling=0.,transfer=0.,projection=0.,apply=0.,audit=0.,learning=0.)
    stamp=time.perf_counter();e.reset();times['reset']=time.perf_counter()-stamp
    t=time.perf_counter();steps=0;attempts=0;updates=0;start_quality=float(e.quality.cpu());reward_sum=0.
    while not e.terminal:
        t0=time.perf_counter();a=p.choose(e,training);times['choose']+=time.perf_counter()-t0;attempts+=p.generation_attempts
        for name,attr in [('model','model_seconds'),('sampling','sampling_seconds'),('transfer','transfer_seconds'),('projection','projection_seconds')]:times[name]+=getattr(p,attr,0.)
        if a is None:
            e.termination=p.last_rejection
            if training and p.tape:p.record(action_reward(e,e.far_error))
            break
        before=e.far_error;t0=time.perf_counter();e.apply(a,audit=False);sync(e.device);times['apply']+=time.perf_counter()-t0;steps+=1
        reward=action_reward(e,before)
        if training:p.record(reward)
        reward_sum+=reward
        if e.edits%8==0 or e.terminal:
            old=float(e.quality.cpu());t0=time.perf_counter();e.evaluate();times['audit']+=time.perf_counter()-t0
            bonus=.1*(float(e.quality.cpu())-old)
            if training:p.add_reward(bonus)
            reward_sum+=bonus
        if training and (e.terminal or len(p.tape)>=32):
            t0=time.perf_counter();p.finish(p.bootstrap(e));sync(e.device);times['learning']+=time.perf_counter()-t0;updates+=1
    if training and p.tape:
        t0=time.perf_counter();p.finish(p.bootstrap(e));sync(e.device);times['learning']+=time.perf_counter()-t0;updates+=1
    # Match Session.finish_episode: failure gets an audit before the final update.
    # Normal success has already been audited exactly once.
    if e.last_audit_edit!=e.edits:
        t0=time.perf_counter();e.evaluate();times['audit']+=time.perf_counter()-t0
    seconds=time.perf_counter()-stamp
    return dict(seconds=seconds,steps=steps,attempts=attempts,updates=updates,success=e.termination=='far_reached',termination=e.termination,
        quality=float(e.quality.cpu()),qualityChange=float(e.quality.cpu())-start_quality,rewardSum=reward_sum,times=times)

def main():
    ap=argparse.ArgumentParser();ap.add_argument('--episodes',type=int,default=5);ap.add_argument('--tag',default='baseline');ap.add_argument('--precompute',action='store_true');args=ap.parse_args();torch.set_num_threads(1)
    path=ROOT/f'results/training-profile-{args.tag}-2026-11-09.json'
    r=dict(platform=platform.platform(),torch=torch.__version__,cpuThreads=1,rows=[],note='Isolated headless profile, including backpropagation and episode reset; not a convergence study.',hashes={str(p.relative_to(ROOT)):hashlib.sha256(p.read_bytes()).hexdigest() for p in [Path(__file__),*sorted((ROOT/'voxelform').glob('*.py'))]})
    context=ContextLibrary().load('m_bcn_0093')
    c=PatchConfig(allocation='feedback',eligibility='depth',budget_reserve_fraction=.2,minimum_patch=4,maximum_patches=4,maximum_voids=1,minimum_void_area=4,batched=True,vertical_interior=True)
    for device in ['cpu']+(['mps'] if torch.backends.mps.is_available() else []):
        for tag,shape,suns,ctx in [('L',(4,48,64),8,None),('XL',(8,64,96),12,None),('Barcelona',context['shape'],8,context)]:
            stamp=time.perf_counter();e=Environment(shape,context=ctx,sun_count=suns,device=device,frontier_factory=functools.partial(PatchFacades,config=c,depth_device=device if device!='cpu' else None));setup=time.perf_counter()-stamp
            p=PatchLearner(device,seed=123,config=c)
            preparation=e.precompute_external() if args.precompute else None
            warm=episode(e,p,61000,True)
            trials=[episode(e,p,61001+i,True) for i in range(args.episodes)]
            stamp=time.perf_counter();buffer=io.BytesIO();torch.save(p.state(),buffer);sync(device);save=time.perf_counter()-stamp
            row=dict(device=device,case=tag,shape=shape,sunSamples=suns,contextBuildings=len(ctx['buildings']) if ctx else 16,setupSeconds=setup,coldTraining=warm,trials=trials,
                externalPreparation=preparation,checkpointSerializationSeconds=save,checkpointBytes=buffer.tell(),rssHighWaterBytes=resource.getrusage(resource.RUSAGE_SELF).ru_maxrss,
                mpsAllocatedBytes=torch.mps.current_allocated_memory() if device=='mps' else None)
            r['rows'].append(row);path.write_text(json.dumps(r,indent=2)+'\n')
            print(args.tag,device,tag,'setup',round(setup,2),'episode',round(float(np.median([x['seconds'] for x in trials])),3),'steps',[x['steps'] for x in trials],'success',sum(x['success'] for x in trials),flush=True)
            del e,p;gc.collect()
            if device=='mps':torch.mps.empty_cache()
    print(path,flush=True)

if __name__=='__main__':main()
